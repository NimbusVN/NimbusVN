# ---------------------------------------------------------------------------
# openartemis 内核（engine/openartemis 子模块）的构建接线。
#
# 子模块内部按 krkrsdl3 的风格收敛为 core/ + environ/（不再自带
# 顶层 CMakeLists / android / web / cmake / script），构建与平台壳全部由
# 超工程负责：根 CMakeLists 统合两个引擎，平台壳在 platform/ 下。
#
# 目标：
#   libopenartemis  — 引擎静态库（core/，C++20）
#   openartemis     — 宿主（桌面/WASM 可执行文件；Android 为 .so）
#   lua51           — vendored Lua 5.1.5
# ---------------------------------------------------------------------------

endif()

    target_compile_definitions(libopenartemis PUBLIC OA_HAVE_FFMPEG=1)
    target_include_directories(libopenartemis SYSTEM PRIVATE ${FFMPEG_INCLUDE_DIRS})
    target_link_directories(libopenartemis PRIVATE ${FFMPEG_LIBRARY_DIRS})
    nimbus_link_ffmpeg(libopenartemis)
    message(STATUS "openartemis: FFmpeg video backend ENABLED (${FFMPEG_LIBRARIES})")

    message(STATUS "openartemis: FFmpeg video backend DISABLED (not installed) — Ogg/Theora only")

# 宿主：桌面/WASM 为可执行文件，Android 为 libopenartemis.so
# ---------------------------------------------------------------------------
if(EMSCRIPTEN)
    set(OA_HOST_PLATFORM ${OA_ROOT}/environ/platform/platform_wasm.cpp)
elseif(ANDROID)
    set(OA_HOST_PLATFORM ${OA_ROOT}/environ/platform/platform_android.cpp)
else()
    set(OA_HOST_PLATFORM ${OA_ROOT}/environ/platform/platform_desktop.cpp)
endif()

if(ANDROID)
    add_library(openartemis SHARED
        ${OA_ROOT}/environ/main.cpp
        ${OA_HOST_PLATFORM})
    set_target_properties(openartemis PROPERTIES OUTPUT_NAME "openartemis")
else()
    add_executable(openartemis
        ${OA_ROOT}/environ/main.cpp
        ${OA_HOST_PLATFORM})
endif()
target_include_directories(openartemis PRIVATE ${OA_ROOT}/environ)
target_link_libraries(openartemis PRIVATE libopenartemis SDL3::SDL3)
set_target_properties(openartemis PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON)

if(EMSCRIPTEN)
    # 浏览器壳：IDBFS 存档挂载、MEMFS 大包、pthreads + 异常。
    # 与 krkrsdl3 的 index.html 分目录输出。
    set_target_properties(openartemis PROPERTIES
        SUFFIX ".html"
        OUTPUT_NAME "index"
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/openartemis")
    target_link_options(openartemis PRIVATE
        -lidbfs.js
        -sFORCE_FILESYSTEM=1
        -sALLOW_MEMORY_GROWTH=1
        -sNO_EXIT_RUNTIME=1
        -sINVOKE_RUN=0
        -sSTACK_SIZE=4MB
        -sEXPORTED_RUNTIME_METHODS=FS,callMain,ccall,cwrap,UTF8ToString
        --shell-file "${CMAKE_SOURCE_DIR}/platform/emscripten/openartemis_shell.html")
    set_property(TARGET openartemis APPEND PROPERTY
        LINK_DEPENDS "${CMAKE_SOURCE_DIR}/platform/emscripten/openartemis_shell.html")
    target_compile_options(openartemis PRIVATE -fexceptions)
    target_link_options(openartemis PRIVATE -fexceptions -sDISABLE_EXCEPTION_CATCHING=0)
    target_link_options(openartemis PRIVATE -sPTHREAD_POOL_SIZE=6)
elseif(ANDROID)
    target_link_libraries(openartemis PRIVATE EGL GLESv3)
    target_link_options(openartemis PRIVATE $<$<CONFIG:Release>:-s>)
endif()

if(MSVC)
    target_compile_options(openartemis PRIVATE /W4)
else()
    target_compile_options(openartemis PRIVATE -Wall -Wextra -Wpedantic)

# ---------------------------------------------------------------------------
# 模糊测试（仅桌面；独立开关，通常不和全量单测一起开）
# ---------------------------------------------------------------------------
if(OA_ENABLE_FUZZ AND NOT ANDROID AND NOT EMSCRIPTEN)
    set(OA_FUZZ_LIBFUZZER_OK OFF)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        set(OA_FUZZ_LIBFUZZER_RUNTIME "" CACHE FILEPATH
            "libFuzzer runtime archive (empty = auto-detect from clang -print-resource-dir)")
        if(NOT OA_FUZZ_LIBFUZZER_RUNTIME)
            execute_process(COMMAND ${CMAKE_CXX_COMPILER} -print-resource-dir
                            OUTPUT_VARIABLE _oa_clang_rt
                            OUTPUT_STRIP_TRAILING_WHITESPACE
                            RESULT_VARIABLE _oa_clang_rt_rc)
            if(_oa_clang_rt_rc EQUAL 0)
                if(CMAKE_SIZEOF_VOID_P EQUAL 8)
                    set(_oa_fuzzer_arch x86_64)
                else()
                    set(_oa_fuzzer_arch i386)
                endif()
                foreach(_pat
                        "${_oa_clang_rt}/lib/windows/libclang_rt.fuzzer-${_oa_fuzzer_arch}.a"
                        "${_oa_clang_rt}/lib/windows/libclang_rt.fuzzer-*.a"
                        "${_oa_clang_rt}/lib/*/libclang_rt.fuzzer-*.a")
                    if(OA_FUZZ_LIBFUZZER_RUNTIME)
                        break()
                    endif()
                    file(GLOB _oa_hit "${_pat}")
                    if(_oa_hit)
                        list(GET _oa_hit 0 OA_FUZZ_LIBFUZZER_RUNTIME)
                    endif()
                endforeach()
            endif()
        endif()
        if(OA_FUZZ_LIBFUZZER_RUNTIME AND EXISTS "${OA_FUZZ_LIBFUZZER_RUNTIME}")
            set(OA_FUZZ_LIBFUZZER_OK ON)
            message(STATUS "openartemis: fuzz targets = libFuzzer (${OA_FUZZ_LIBFUZZER_RUNTIME})")
        else()
            message(STATUS "openartemis: fuzz targets = brute mode (no libFuzzer runtime)")
        endif()
    endif()
    add_subdirectory(${OA_ROOT}/tests/fuzz ${CMAKE_BINARY_DIR}/openartemis-fuzz)
endif()
